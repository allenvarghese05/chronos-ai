# T0.5 Plan: Renderer scaffold `packages/renderer`

## 1. Summary

Creates `packages/renderer/` — a fully configured Remotion 4.0.534 / React 19.3.0 project that
can produce the "Morning Reel" 1080×1920 MP4 from a JSON props file via `npx remotion render`.
`MorningReel` composition (30 fps, `calculateMetadata` duration) wraps a working `CoverScene` plus
typed stubs for `StoryScene`, `OutroScene`, and `WordCaptions`. Hand-written zod `ReelProps` schema,
vitest tests with explicit expected values for timing and caption helpers, Biome + TypeScript strict,
and a CI smoke still at frame 60 all land in this task.

## 2. Design

**Approach:** Standalone pnpm workspace in `packages/renderer/` (isolated from `apps/mobile`).
`remotion.config.ts` sets 1080×1920 30 fps defaults. `calculateMetadata` derives
`durationInFrames = ceil(audio.durationSec × fps) + 15` from `ReelProps`. Scenes are rendered
inside `<Sequence>` using `sceneFrames()` from `src/lib/timing.ts`. Design tokens (`tokens.ts`)
centralise all typographic, colour, spacing and motion values. Fonts are self-hosted OFL files
under `public/fonts/` loaded at render time via `@remotion/fonts`. The zod schema (`src/schema.ts`)
is hand-written now and will be replaced by generated code in T3.4.

**Alternatives rejected:**
- `@remotion/google-fonts` at render time: adds network dependency at 05:30 AM; self-host is safer.
- Snapshot tests for timing/captions: brief explicitly requires explicit expected values, not snapshots.
- Minimal props schema: brief/owner require full `ReelProps` per docs/07 §4.
- `nodeLinker: hoisted` for renderer: not needed (Remotion doesn't use Metro), but included for
  monorepo consistency (matches mobile convention).

```typescript
export const CaptionWordSchema = z.object({
  w: z.string().min(1),
  s: z.number().nonnegative(),
  e: z.number().nonnegative(),

const SourceSchema = z.object({ name: z.string(), url: z.string().url() });

    index: z.number().int().nonnegative(),

    media: MediaSchema.optional(),
    source: SourceSchema.optional(),
    startSec: z.number().nonnegative(),
    endSec: z.number().nonnegative(),
  }).refine(({ startSec, endSec }) => endSec >= startSec, {
    message: "endSec must be >= startSec",
  }),
  z.object({
    index: z.number().int().nonnegative(),
    type: z.literal("story"),
    title: z.string().optional(),
    body: z.array(z.string()).optional(),
    media: MediaSchema.optional(),
    source: SourceSchema.optional(),
    startSec: z.number().nonnegative(),
    endSec: z.number().nonnegative(),
  }).refine(({ startSec, endSec }) => endSec >= startSec, {
    message: "endSec must be >= startSec",
  }),
  z.object({
    index: z.number().int().nonnegative(),
    type: z.literal("outro"),
    title: z.string().optional(),
    body: z.array(z.string()).optional(),
    media: MediaSchema.optional(),
    source: SourceSchema.optional(),
    startSec: z.number().nonnegative(),
    endSec: z.number().nonnegative(),
  }).refine(({ startSec, endSec }) => endSec >= startSec, {
    message: "endSec must be >= startSec",
  }),
]);

export const ReelPropsSchema = z.object({
  localDate: z.string().regex(/^\d{4}-\d{2}-\d{2}$/, "must be YYYY-MM-DD"),
  greeting: z.string().min(1),
  tone: z.enum(["friendly", "formal"]),
  audio: z.object({
    src: z.string().min(1),
    durationSec: z.number().positive(),

  captions: z.array(z.array(CaptionWordSchema)),
  theme: z.enum(["dawn", "mono", "ink"]),
  attribution: z.array(z.string()),
});

export type CaptionWord = z.infer<typeof CaptionWordSchema>;
export type SceneProps = z.infer<typeof ScenePropsSchema>;
export type ReelProps = z.infer<typeof ReelPropsSchema>;
```

```typescript
/**
 * sceneFrames — convert a scene's [startSec, endSec] into [startFrame, durationInFrames].
 *
 * Rounding: startFrame = Math.round(startSec * fps),
 *           endFrame   = Math.round(endSec   * fps),
 *           durationInFrames = max(0, endFrame - startFrame).
 *
 * A zero-duration scene (startSec === endSec) returns durationInFrames = 0
 * and is safe to pass to <Sequence> (it renders nothing).
 */
export function sceneFrames(
  startSec: number,
  endSec: number,
  fps: number,
): { startFrame: number; durationInFrames: number } {
  const startFrame = Math.round(startSec * fps);
  const endFrame = Math.round(endSec * fps);
  const durationInFrames = Math.max(0, endFrame - startFrame);
  return { startFrame, durationInFrames };
}
```

**`src/lib/captions.ts`:**
```typescript
import type { CaptionWord } from "../schema.js";

export interface CaptionLine {
  words: CaptionWord[];
}

export interface LayoutOptions {
  minPerLine: number;
  maxPerLine: number;
}

/**
 * layoutCaptions — group an array of CaptionWords into display lines of
 * minPerLine..maxPerLine words. The last line must have ≥ minPerLine words
 * (no orphan single-word last line). If the total word count is less than
 * minPerLine, all words are placed on one line.
 */
export function layoutCaptions(
  words: CaptionWord[],
  { minPerLine = 3, maxPerLine = 5 }: Partial<LayoutOptions> = {},
): CaptionLine[] {
  if (words.length === 0) return [];
  if (words.length <= minPerLine) return [{ words }];

  const lines: CaptionLine[] = [];
  let i = 0;
  while (i < words.length) {
    const remaining = words.length - i;
    // If only enough words for one last line without orphan, take them all
    if (remaining <= maxPerLine) {
      // Check: if remaining < minPerLine, the previous line should have absorbed some words
      // but we can only fix that if lines.length > 0
      if (remaining < minPerLine && lines.length > 0) {
        // Merge into previous line (no orphan)
        const prev = lines[lines.length - 1];
        prev.words = [...prev.words, ...words.slice(i)];
        break;
      }
      lines.push({ words: words.slice(i) });
      break;
    }
    lines.push({ words: words.slice(i, i + maxPerLine) });
    i += maxPerLine;
  }
  return lines;
}
```

**timing.test.ts** — explicit tests:
- `sceneFrames(0, 1, 30)` → `{ startFrame: 0, durationInFrames: 30 }`
- `sceneFrames(1, 2, 30)` → `{ startFrame: 30, durationInFrames: 30 }`
- `sceneFrames(0, 0, 30)` → `{ startFrame: 0, durationInFrames: 0 }` (zero duration)
- `sceneFrames(0, 0.5, 30)` → `{ startFrame: 0, durationInFrames: 15 }`
- `sceneFrames(0, 1/3, 30)` → `{ startFrame: 0, durationInFrames: 10 }` (fractional, rounding)
- `sceneFrames(0, 3.0, 30)` → `{ startFrame: 0, durationInFrames: 90 }`

**captions.test.ts** — explicit tests:
- empty input → `[]`
- 1 word → `[{ words: [word1] }]` (< minPerLine → one line)
- 2 words → `[{ words: [w1, w2] }]`
- 3 words → `[{ words: [w1, w2, w3] }]`
- 5 words → `[{ words: [w1..w5] }]`
- 6 words → `[{ words: [w1..w5] }, { words: [w6] }]` — BUT w6 < minPerLine(3) so merge → `[{ words: [w1..w6] }]`
- 8 words → `[{ words: [w1..w5] }, { words: [w6..w8] }]` (remaining 3 = minPerLine)
- 10 words → `[{ words: [w1..w5] }, { words: [w6..w10] }]`
- 11 words → `[{ words: [w1..w5] }, { words: [w6..w10] }, { words: [w11] }]` — w11 < min → merge: `[{ words: [w1..w5] }, { words: [w6..w11] }]`
