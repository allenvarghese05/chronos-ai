# T2.1 Plan: Swift module `chronos-sensors`

## 1. Summary

When done, `apps/mobile/modules/chronos-sensors` contains a fully implemented Expo native module
for iOS that exposes CoreMotion (steps, activity summary), UIKit device state, and BGTaskScheduler
background-refresh registration to JavaScript.  A pure-Swift SwiftPM package `ChronosSensorsCore`
hosts the testable business logic (`ActivitySummariser`, `DeviceStateMapper`, protocols,
constants) and is tested with `swift test`.  The thin Expo module wrapper bridges CoreMotion
adapters to those pure types and emits `backgroundRefresh` events via an `AppDelegateSubscriber`.
The TypeScript surface, Jest mocks, and the index screen are updated to match.  `just mobile-ios-test`
and `just check-mobile` both pass; the Simulator shows `deviceState()` JSON and the
`pedometer_unavailable` path.

## 2. Design

**SwiftPM package inside the module** (`ios/ChronosSensorsCore/`): pure Swift, zero ExpoModulesCore
imports, tested standalone with `swift test`.  The podspec adds the Core Sources to the compile
with `s.source_files = ["**/*.swift", "ChronosSensorsCore/Sources/**/*.swift"]` excluding the
test directory.

**Activity duration algorithm**: sort samples by `startDate`; for each sample compute the duration
as `min(next.startDate, to) - max(sample.startDate, from)` clamped to ≥ 0; the activity type is
chosen by precedence (automotive > cycling > running > walking > stationary > unknown) among the
sample's true flags; `walkingConf` = max confidence over all walking-or-running samples.

**`deviceState()` as `Function`** (synchronous) using `@MainActor` dispatch on the Expo event
loop (Expo `Function` closures run on the caller's thread; wrap UIKit reads in `DispatchQueue.main.sync`).

**AppDelegateSubscriber** registers the BGTaskScheduler handler at app launch (before
`applicationDidFinishLaunching` returns) to satisfy iOS's requirement; a 25-second auto-complete
timer prevents the watchdog kill if JS never calls `completeBackgroundRefresh`.

**Rejected alternatives**:
- Option B (Xcode test target): requires Xcode workspace + prebuild; SwiftPM is simpler.
- Passing identifier from JS only: identifier is a constant in both places; a Jest test asserts equality.

```swift
import Foundation

/// Mirrors the fields of CMMotionActivity that we consume.
/// Pure value type so it can be constructed in tests without CoreMotion.
public struct ActivitySample: Sendable {
    public let startDate: Date
    public let stationary: Bool
    public let walking: Bool
    public let running: Bool
    public let automotive: Bool
    public let cycling: Bool
    public let unknown: Bool
    /// 0=low 1=medium 2=high (mirrors CMMotionActivityConfidence raw value)
    public let confidence: Int

    public init(startDate: Date, stationary: Bool = false, walking: Bool = false,
                running: Bool = false, automotive: Bool = false,
                cycling: Bool = false, unknown: Bool = false, confidence: Int = 0) {
        self.startDate = startDate; self.stationary = stationary
        self.walking = walking; self.running = running
        self.automotive = automotive; self.cycling = cycling
        self.unknown = unknown; self.confidence = confidence
    }

    /// Dominant activity type by precedence: automotive>cycling>running>walking>stationary>unknown.
    public var dominantType: ActivityType {
        if automotive { return .automotive }
        if cycling    { return .cycling }
        if running    { return .running }
        if walking    { return .walking }
        if stationary { return .stationary }
        return .unknown
    }

    public enum ActivityType { case automotive, cycling, running, walking, stationary, unknown }
}
```

**ActivitySummariser.swift** (exact complete source):
```swift
import Foundation

/// Summary of activity durations over a query window.
public struct ActivitySummary: Sendable {
    public let walkingSec: Int
    public let runningSec: Int
    public let stationarySec: Int
    public let automotiveSec: Int
    public let cyclingSec: Int
    public let unknownSec: Int
    /// Max confidence over walking+running samples ("low"|"medium"|"high").
    public let walkingConf: String
}

/// Error thrown when the query range is invalid.
public enum SummariserError: Error, Sendable {
    case invalidRange(String)
}

/// Pure function: maps an array of change-point ActivitySamples to durations.
/// - Parameters:
///   - samples: Unsorted; may start before `from` (clamped).
///   - from: Inclusive start of the query window.
///   - to: Exclusive end of the query window (clamped to from+7days max).
public func summarise(samples: [ActivitySample], from: Date, to: Date) throws -> ActivitySummary {
    guard from <= to else {
        throw SummariserError.invalidRange("from (\(from)) must be ≤ to (\(to))")
    }

    // Clamp to 7-day CoreMotion limit.
    let sevenDays: TimeInterval = 7 * 24 * 3600
    let effectiveTo = min(to, from.addingTimeInterval(sevenDays))

    // Sort ascending; keep only samples that start before effectiveTo.
    let sorted = samples
        .filter { $0.startDate < effectiveTo }
        .sorted { $0.startDate < $1.startDate }

    var walkingSec   = 0
    var runningSec   = 0
    var stationarySec = 0
    var automotiveSec = 0
    var cyclingSec   = 0
    var unknownSec   = 0
    var maxWalkingConf = -1   // -1 = no walking/running sample seen

    for (i, sample) in sorted.enumerated() {
        let segStart = max(sample.startDate, from)
        let nextStart: Date
        if i + 1 < sorted.count {
            nextStart = sorted[i + 1].startDate
        } else {
            nextStart = effectiveTo
        }
        let segEnd = min(nextStart, effectiveTo)
        let duration = max(0.0, segEnd.timeIntervalSince(segStart))
        let secs = Int(duration.rounded())

        switch sample.dominantType {
        case .walking:    walkingSec    += secs
        case .running:    runningSec    += secs
        case .stationary: stationarySec += secs
        case .automotive: automotiveSec += secs
        case .cycling:    cyclingSec    += secs
        case .unknown:    unknownSec    += secs
        }

        // Track max confidence for walking+running samples.
        if sample.walking || sample.running {
            maxWalkingConf = max(maxWalkingConf, sample.confidence)
        }
    }

    let walkingConf: String
    switch maxWalkingConf {
    case 2:  walkingConf = "high"
    case 1:  walkingConf = "medium"
    case 0:  walkingConf = "low"
    default: walkingConf = "low"   // no walking/running sample
    }

    return ActivitySummary(
        walkingSec: walkingSec, runningSec: runningSec,
        stationarySec: stationarySec, automotiveSec: automotiveSec,
        cyclingSec: cyclingSec, unknownSec: unknownSec,
        walkingConf: walkingConf
    )
}
```

### Step 3 — `DeviceStateMapper`

**File:** `DeviceStateMapper.swift`

```swift
import Foundation

/// Raw inputs (all primitives) so the mapper is testable without UIKit.
public struct DeviceStateInputs: Sendable {
    public let isProtectedDataAvailable: Bool
    /// UIDevice.batteryState raw value: 0=unknown,1=unplugged,2=charging,3=full
    public let batteryStateRaw: Int
    /// UIDevice.batteryLevel: [0,1] or -1 if unavailable
    public let batteryLevel: Float
    /// UIApplication.applicationState raw value: 0=active,1=inactive,2=background
    public let appStateRaw: Int
    public let isLowPowerModeEnabled: Bool
    public let iosVersion: String

    public init(isProtectedDataAvailable: Bool, batteryStateRaw: Int,
                batteryLevel: Float, appStateRaw: Int,
                isLowPowerModeEnabled: Bool, iosVersion: String) {
        self.isProtectedDataAvailable = isProtectedDataAvailable
        self.batteryStateRaw = batteryStateRaw
        self.batteryLevel = batteryLevel
        self.appStateRaw = appStateRaw
        self.isLowPowerModeEnabled = isLowPowerModeEnabled
        self.iosVersion = iosVersion
    }
}

/// Pure mapping → string-keyed dictionary ready for JSON serialisation.
public func mapDeviceState(_ inputs: DeviceStateInputs) -> [String: Any?] {
    let batteryState: String
    switch inputs.batteryStateRaw {
    case 1:  batteryState = "unplugged"
    case 2:  batteryState = "charging"
    case 3:  batteryState = "full"
    default: batteryState = "unknown"
    }

    let batteryLevel: Double? = inputs.batteryLevel < 0 ? nil : Double(inputs.batteryLevel)

    let appState: String
    switch inputs.appStateRaw {
    case 0:  appState = "active"
    case 1:  appState = "inactive"
    default: appState = "background"
    }

    return [
        "isUnlocked":    inputs.isProtectedDataAvailable,
        "batteryState":  batteryState,
        "batteryLevel":  batteryLevel as Any?,
        "appState":      appState,
        "lowPowerMode":  inputs.isLowPowerModeEnabled,
        "iosVersion":    inputs.iosVersion,
    ]
}
```
