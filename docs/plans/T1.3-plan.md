# T1.3 Plan: Window and local-day computation

## 1. Summary

Creates `services/api/src/chronos/wake/window.py` (and `__init__.py`) with two frozen Pydantic
models — `WindowConfig` and `Window` — plus six pure functions: `local_date_for`,
`day_kind`, `window_for`, `is_in_window`, `next_window_start`, and `tick_instants`.
All functions are timezone-aware, DST-safe, have no I/O and no DB access.
Tests in `tests/wake/test_window.py` cover all 17 concrete cases and three Hypothesis
property checks; 100 % branch and line coverage of the module is enforced.

## 2. Design

**WindowConfig** is a frozen Pydantic v2 model (`extra="forbid"`) so T1.4's YAML loader can
produce it via `model_validate`. Fields use stdlib `datetime.time` / `datetime.date`.
`holidays: frozenset[date]` (not a set) so the model stays hashable and Pydantic doesn't mutate it.

**Window** is also a frozen Pydantic model; its `start_utc` / `end_utc` are always UTC-aware,
`start_local` / `end_local` are naive `datetime.time` values (the actual resolved wall-clock
times after DST adjustment).

**DST gap (spring-forward)**: `_resolve_local_to_utc` detects a gap by comparing the fold=0
UTC round-trip local time against the configured local time. When they differ, a binary search
between the fold=0 and fold=1 UTC values narrows to the exact second of the transition. This
gives the first valid UTC in the post-gap period (e.g., 07:00:00 Z = 03:00 AM EDT for the
NY spring-forward). The returned `Window.start_local` then reflects the shifted wall-clock
time, not the configured time.

**Ambiguous time (fall-back)**: fold=0 is the default for `datetime.datetime()`, so the
first occurrence (earlier UTC) is used automatically; no extra handling needed.

**`next_window_start(after)`**: iterates local dates forward from `local_date_for(after, tz,
boundary)` until it finds a window whose `start_utc > after` (strictly). An in-progress window
is never returned (owner answer 5).

Rejected alternatives:
- Frozen dataclass instead of Pydantic: breaks T1.4's `model_validate` round-trip and loses
  field validators.
- Shifting gap times by `timedelta(hours=gap_size)` only: gives 3:30 AM instead of 3:00 AM
  for NY spring-forward (wrong; see brief §open-question-4).
- Using `zoneinfo` private transition tables: internal API; binary search is robust and fast
  (≤14 iterations over utc0−utc1 range, always ≤1 hour for any real DST rule).

```python
def local_date_for(
    instant: datetime.datetime,
    tz: ZoneInfo,
    boundary: datetime.time,
) -> datetime.date:
    """Return the local calendar date for a UTC instant using a custom day boundary.

    The "local date" rolls over at *boundary* local time (not midnight).  An instant
    before *boundary* on calendar date D is attributed to date D-1.

    Args:
        instant: A UTC-aware datetime (tzinfo must not be None).
        tz: IANA timezone (ZoneInfo object).
        boundary: Local wall-clock time at which a new day begins (naive).

    Returns:
        A ``datetime.date`` in the local calendar.

    Raises:
        ValueError: If *instant* is naive (tzinfo is None).
    """
    if instant.tzinfo is None:
        raise ValueError("instant must be timezone-aware; got a naive datetime")
    local_dt = instant.astimezone(tz)
    if local_dt.time().replace(tzinfo=None) < boundary:
        return local_dt.date() - datetime.timedelta(days=1)
    return local_dt.date()


def day_kind(local_date: datetime.date, cfg: WindowConfig) -> DayKind:
    """Classify a local date as weekday, weekend, or holiday.

    Rules (in priority order):
    1. Saturday (weekday()==5) or Sunday (weekday()==6) → ``DayKind.weekend``.
    2. Date in ``cfg.holidays`` → ``DayKind.holiday`` (uses weekend window times).
    3. All others → ``DayKind.weekday``.

    A holiday that falls on a Saturday/Sunday is still ``DayKind.weekend`` (rule 1 wins).

    Args:
        local_date: The local calendar date to classify.
        cfg: Window configuration with the holidays set.

    Returns:
        ``DayKind`` enum member.
    """
    if local_date.weekday() >= 5:
        return DayKind.weekend
    if local_date in cfg.holidays:
        return DayKind.holiday
    return DayKind.weekday
```

```python
def _resolve_local_to_utc(
    date: datetime.date,
    t: datetime.time,
    tz: ZoneInfo,
) -> datetime.datetime:
    """Convert a local (date, time) pair to a UTC-aware datetime.

    Handles:
    - Normal times: standard UTC conversion.
    - Spring-forward gap: shifts to the *first* valid UTC second after the gap.
    - Fall-back ambiguity: uses fold=0 (first occurrence, earlier UTC).

    Returns a UTC-aware datetime with microsecond=0.
    """
    # Build with fold=0 (first occurrence / pre-transition).
    local_fold0 = datetime.datetime(
        date.year, date.month, date.day,
        t.hour, t.minute, t.second, 0,
        fold=0, tzinfo=tz,
    )
    utc0 = local_fold0.astimezone(datetime.UTC)
    # Detect gap: if UTC round-trip gives a different local time, we're in a gap.
    roundtrip = utc0.astimezone(tz)
    if roundtrip.replace(tzinfo=None) != local_fold0.replace(tzinfo=None):
        # Spring-forward gap: find exact transition using fold=1 as lower bound.
        local_fold1 = datetime.datetime(
            date.year, date.month, date.day,
            t.hour, t.minute, t.second, 0,
            fold=1, tzinfo=tz,
        )
        utc1 = local_fold1.astimezone(datetime.UTC)
        # utc1 < transition_utc ≤ utc0; binary-search to 1-second precision.
        utc0 = _find_transition_utc(utc1, utc0, tz)
    return utc0


def _find_transition_utc(
    lo: datetime.datetime,
    hi: datetime.datetime,
    tz: ZoneInfo,
) -> datetime.datetime:
    """Binary-search for the first UTC second in the post-transition period.

    *lo* must be pre-transition, *hi* must be post-transition.
    Returns a UTC datetime with microsecond=0 that is the exact transition second.
    At most ceil(log2((hi-lo)/1s)) ≈ 14 iterations for a 3-hour window.
    """
    post_offset: datetime.timedelta | None = hi.astimezone(tz).utcoffset()
    while (hi - lo).total_seconds() > 1.0:
        mid = lo + (hi - lo) / 2
        if mid.astimezone(tz).utcoffset() == post_offset:
            hi = mid
        else:
            lo = mid
    # Snap to second boundary; hi is in [T, T+1) where T is transition second.
    result = hi.replace(microsecond=0)
    if result.astimezone(tz).utcoffset() != post_offset:
        result += datetime.timedelta(seconds=1)
    return result


def window_for(
    local_date: datetime.date,
    tz: ZoneInfo,
    cfg: WindowConfig,
) -> Window:
    """Compute the UTC window for *local_date* in *tz* according to *cfg*.

    DST gaps shift the window boundary forward; ambiguous times use fold=0.
    Returns a ``Window`` with UTC-aware start/end and naive local times.

    Args:
        local_date: The local calendar date.
        tz: IANA timezone.
        cfg: Window configuration.

    Returns:
        ``Window`` with start/end in both UTC and local wall-clock time.
    """
    kind = day_kind(local_date, cfg)
    if kind in (DayKind.weekend, DayKind.holiday):
        start_t, end_t = cfg.weekend_start, cfg.weekend_end
    else:
        start_t, end_t = cfg.weekday_start, cfg.weekday_end

    start_utc = _resolve_local_to_utc(local_date, start_t, tz)
    end_utc = _resolve_local_to_utc(local_date, end_t, tz)

    return Window(
        local_date=local_date,
        kind=kind,
        start_utc=start_utc,
        end_utc=end_utc,
        start_local=start_utc.astimezone(tz).time().replace(tzinfo=None),
        end_local=end_utc.astimezone(tz).time().replace(tzinfo=None),
    )


def is_in_window(instant: datetime.datetime, window: Window) -> bool:
    """Return True iff *instant* ∈ [window.start_utc, window.end_utc).

    Args:
        instant: A UTC-aware datetime.
        window: A computed Window.

    Returns:
        True if start_utc ≤ instant < end_utc.

    Raises:
        ValueError: If *instant* is naive.
    """
    if instant.tzinfo is None:
        raise ValueError("instant must be timezone-aware")
    return window.start_utc <= instant < window.end_utc
```

```python
def next_window_start(
    after: datetime.datetime,
    tz: ZoneInfo,
    cfg: WindowConfig,
) -> datetime.datetime:
    """Return the UTC start of the next window whose start is strictly after *after*.

    An in-progress window (already started) is not returned.

    Args:
        after: A UTC-aware datetime reference point.
        tz: IANA timezone.
        cfg: Window configuration.

    Returns:
        UTC datetime of the next window start, strictly > *after*.

    Raises:
        ValueError: If *after* is naive.
    """
    if after.tzinfo is None:
        raise ValueError("after must be timezone-aware")
    check_date = local_date_for(after, tz, cfg.day_boundary)
    for _ in range(400):  # safety limit; searches at most ~1 year forward
        window = window_for(check_date, tz, cfg)
        if window.start_utc > after:
            return window.start_utc
        check_date += datetime.timedelta(days=1)
    raise RuntimeError("next_window_start: no window found within 400 days")  # pragma: no cover


def tick_instants(window: Window, interval_min: int) -> list[datetime.datetime]:
    """Generate UTC tick timestamps at *interval_min*-minute intervals within *window*.

    Ticks start at window.start_utc (inclusive) and stop before window.end_utc (exclusive).
    All ticks are UTC-aware and equally spaced.

    Args:
        window: The time window (must have start_utc < end_utc).
        interval_min: Interval in minutes (positive integer).

    Returns:
        Ordered list of UTC datetimes; may be empty if interval_min ≥ window duration.
    """
    ticks: list[datetime.datetime] = []
    t = window.start_utc
    delta = datetime.timedelta(minutes=interval_min)
    while t < window.end_utc:
        ticks.append(t)
        t += delta
    return ticks
```

```python
from hypothesis import given, settings as h_settings
from hypothesis import strategies as st

_ZONES = [ZoneInfo("Asia/Kolkata"), ZoneInfo("America/New_York")]
_tz_st = st.sampled_from(_ZONES)
_date_st = st.dates(min_value=datetime.date(2020, 1, 1), max_value=datetime.date(2030, 12, 31))
_instant_st = st.datetimes(
    min_value=datetime.datetime(2020, 1, 1),
    max_value=datetime.datetime(2030, 12, 31, 23, 59, 59),
    timezones=st.just(datetime.UTC),
)
_FIXED_CFG = WindowConfig(
    weekday_start=datetime.time(6, 0),
    weekday_end=datetime.time(11, 0),
    weekend_start=datetime.time(8, 0),
    weekend_end=datetime.time(14, 0),
)

@given(t1=_instant_st, t2=_instant_st, tz=_tz_st)
@h_settings(deadline=None)
def test_local_date_for_monotone(t1, t2, tz):
    """local_date_for is monotone non-decreasing in the instant."""
    if t1 > t2:
        t1, t2 = t2, t1
    boundary = datetime.time(3, 0)
    d1 = local_date_for(t1, tz, boundary)
    d2 = local_date_for(t2, tz, boundary)
    assert d1 <= d2

@given(local_date=_date_st, tz=_tz_st)
@h_settings(deadline=None)
def test_window_start_before_end_property(local_date, tz):
    """window.start_utc < window.end_utc always holds."""
    window = window_for(local_date, tz, _FIXED_CFG)
    assert window.start_utc < window.end_utc

@given(local_date=_date_st, tz=_tz_st, interval_min=st.integers(min_value=2, max_value=30))
@h_settings(deadline=None)
def test_tick_instants_properties(local_date, tz, interval_min):
    """tick_instants: strictly increasing, all inside [start, end), equally spaced."""
    cfg = WindowConfig(
        weekday_start=datetime.time(6, 0),
        weekday_end=datetime.time(11, 0),
        weekend_start=datetime.time(8, 0),
        weekend_end=datetime.time(14, 0),
        tick_interval_min=interval_min,
    )
    window = window_for(local_date, tz, cfg)
    ticks = tick_instants(window, interval_min)
    delta = datetime.timedelta(minutes=interval_min)
    for tick in ticks:
        assert window.start_utc <= tick < window.end_utc
    for i in range(len(ticks) - 1):
        assert ticks[i + 1] - ticks[i] == delta
```
